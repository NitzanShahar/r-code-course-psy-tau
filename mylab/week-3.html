עבודה עם מבנה נתונים ב

ברוכים הבאים לשבוע השלישי של הקורס!

השבוע נעמיק את היכרותנו עם מבני נתונים, נלמד על סוגים נוספים של אובייקטים, ונשתמש בכלים סטטיסטיים וגרפיים כדי להפיק עוד יותר מידע מהנתונים שלנו.

נתחיל מהיכרות עם פעולות שימושיות על מבני נתונים כמו הדפסת משתנה בודד או תצפית בודדת, הורדת משתנה, הורדת שורות, או שינוי של תא בודד. כל אלה יעזרו לנו לשלוט טוב יותר בטבלאות שלנו ולעשות בהן שינויים מדויקים.

נרחיב גם את ארגז הכלים שלנו עם שני סוגי משתנים חשובים: משתנה לוגי (שיכול לקבל ערך של אמת או שקר), ומשתנה קאטגוריאלי (שמשמש לקבוצות עם קטגוריה שמית). נלמד כיצד ניתן להשתמש במשתנה קטגוריאלי כדי לערוך ולבצע פילטור במבנה הנתונים שלנו.

לאחר מכן נלמד איך לחשב ולהציג סטטיסטיקה תיאורית לפי קבוצה (למשל, ממוצע גובה לפי מין), ואיך לעשות את זה גם בעזרת פילטר, כשאנחנו רוצים להתמקד בתת-קבוצה מהנתונים.

#### GENERAE DATA ----

library(tidyverse)
df = read_csv("data.csv")

# 1. Accessing data frame
print(df$age)
print(df |> slice(5))

# 2. Updateing data frame 
df = df |> 
  select(-depression) |> 
  slice(-c(60, 72))

print(df$age[10])
df$age[10] = 20
print(df$age[10])


# 3. subdivisions and logical 
print(dim(df))

df = df |> 
  mutate(iq_group = iq > median(iq, na.rm = TRUE))

df = df |> 
  mutate(gender = as.factor(gender))
print(levels(df$gender))

df_subset = df |> 
  filter(iq_group == TRUE, gender == "female")


#### DATA SUMMARY ----

# 1. Print descriptive statistics
df |> 
  filter(iq_group == TRUE) |> 
  summarise(
    n = n(),
    age_mean = mean(age, na.rm = TRUE),
    age_sd = sd(age, na.rm = TRUE),
    age_min = min(age, na.rm = TRUE),
    age_max = max(age, na.rm = TRUE)
  )

df |> 
  group_by(gender) |> 
  summarise(
    n = n(),
    age_mean = mean(age, na.rm = TRUE),
    age_sd = sd(age, na.rm = TRUE),
    anxiety_mean = mean(anxiety, na.rm = TRUE),
    anxiety_sd = sd(anxiety, na.rm = TRUE)
  )


# 2. Boxplot
ggplot(df, aes(x = gender, y = anxiety)) +
  geom_boxplot() +
  labs(title = "Anxiety by Gender", x = "Gender", y = "Anxiety")


# 3. Save
write_csv(df, "updated_data.csv")
