)
 
 
 
 
 
ggplot(df, aes(x = group, y = memory_score, color = treatment)) +

חלק א - הצגה גרפית בסיסית:

  1. העלו את קובץ הטיטניק
  2. צרו גרף ממוצעים לסיכויי ההשרדות לפי מחלקה ומגדר. הוסיפו טעויות תקן.
  3. שרטטו את הגיל לפי מחלקה ומגדר באמצעות Box plot / Violin plot
  4. הוסיפו שירטוט נוסף לבחירתכם כדי להדגים את הנתונים
  5. הוסיפו Annotations (למשל: חיצים, טקסט) לפחות באחד הגרפים כדי להדגיש או להסביר מידע חשוב (לדוגמה: ממוצעים או הבדלים משמעותיים בין קבוצות).

חלק ב - שילוב מספר פאנלים בתרשים אחד (Faceting / Subplots):

  1. ארגנו את הגרפים שיצרתם בשלב הקודם לתרשים אחד עם מספר פאנלים.
  2. התאימו את הפריסה (מספר השורות והעמודות) ואת הפרופורציות בין הפאנלים להצגה נוחה של הנתונים, והוסיפו כותרת ראשית לתרשים כולו.

חלק ג - שמירת הגרפים:

  1. שמרו את כל הלוח המשולב בפורמטים הבאים:
    • קובץ תמונה (PNG)
    • קובץ PDF באיכות גבוהה

הקפידו על שמות ברורים לקבצים, וכן על תיעוד בקוד המפרט כל שלב ביצירת הגרפים.

כותרות הקוד: השתמשו בפורמט הכותרות של הקורס, וכתבו בתחילת הקובץ את שם הקורס, השבוע, מספר תעודת הזהות שלכם ושמכם.

#### PART A - BASIC GRAPHICAL PRESENTATION ----

# 1. Load libraries and Titanic data
library(tidyverse)
library(ggdist)
library(patchwork)

df = read_csv("titanic.csv")

# 2. Clean relevant variables
df = df |>
  drop_na(Age, Sex, Pclass, Survived, Embarked)

# 3. Average survival by class and sex with standard error
plot1 = df |>
  group_by(Pclass, Sex) |>
  summarise(
    mean_surv = mean(Survived),
    se_surv   = sd(Survived) / sqrt(n()),
    .groups   = "drop"
  ) |>
  ggplot(aes(x = factor(Pclass), y = mean_surv, fill = Sex)) +
  geom_col(position = position_dodge(width = 0.8)) +
  geom_errorbar(aes(ymin = mean_surv - se_surv,
                    ymax = mean_surv + se_surv),
                position = position_dodge(width = 0.8),
                width = 0.2) +
  labs(x = "Class", y = "Average Survival",
       title = "Average Survival by Class and Sex") +
  annotate("text", x = 1, y = 0.9,
           label = "Females in 1st class\nhad higher survival",
           color = "red", fontface = "bold")

# 4. Age distribution by class and sex (Boxplot)
plot2 = ggplot(df, aes(x = factor(Pclass), y = Age, fill = Sex)) +
  geom_boxplot() +
  labs(x = "Class", y = "Age",
       title = "Age Distribution by Class and Sex")

# 5. Age distribution by survival and sex (Violin using ggdist)
plot3 = ggplot(df, aes(x = factor(Survived), y = Age, fill = Sex)) +
  stat_halfeye(position = position_dodge(width = 0.8)) +
  labs(x = "Survival (0=No, 1=Yes)", y = "Age",
       title = "Age and Survival by Sex (Violin)")

# 6. Additional plot: Histogram of Age by Sex
plot4 = ggplot(df, aes(x = Age, fill = Sex)) +
  geom_histogram(alpha = 0.6, position = "identity", bins = 30) +
  labs(x = "Age", y = "Count",
       title = "Histogram of Age by Sex")


#### PART B - FACETING / SUBPLOTS ----

# 1. Arrange plots in a grid
combined_plot = (plot1 | plot2) /
                (plot3 | plot4) +
  plot_annotation(
    title = "Titanic Data Visualizations",
    theme = theme(plot.title = element_text(size = 16, face = "bold"))
  ) &
  theme_minimal()


#### PART C - SAVING THE COMBINED PLOTS ----

# 1. Save in PNG format
ggsave("titanic_visualizations.png", combined_plot,
       width = 12, height = 10, dpi = 300)

# 2. Save in high-quality PDF format
ggsave("titanic_visualizations.pdf", combined_plot,
       width = 12, height = 10)
